cmake_minimum_required(VERSION 3.16)
project(dataeng LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(dataeng dataeng.cpp)
target_include_directories(dataeng PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
enable_testing()

add_executable(edifact_tests edifact_tests.cpp)
target_link_libraries(edifact_tests PRIVATE dataeng)
add_test(NAME edifact_tests COMMAND edifact_tests)
