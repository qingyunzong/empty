cmake_minimum_required(VERSION 3.16)
project(dataeng LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(dataeng dataeng.cpp)
target_include_directories(dataeng PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
enable_testing()
add_executable(observe_state_test tests/observe_state_test.cpp)
target_link_libraries(observe_state_test PRIVATE dataeng)
add_test(NAME observe_state_test COMMAND observe_state_test)
