cmake_minimum_required(VERSION 3.20)
project(dlks_formal LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(dlks_formal INTERFACE)
target_include_directories(dlks_formal INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(dlks_formal INTERFACE cxx_std_20)

include(CTest)
if(BUILD_TESTING)
  add_executable(paged_cache_tests tests/test_paged_cache.cpp)
  target_link_libraries(paged_cache_tests PRIVATE dlks_formal)
  target_compile_options(paged_cache_tests PRIVATE -Wall -Wextra -O2)
  add_test(NAME paged_cache_tests COMMAND paged_cache_tests)
endif()
