cmake_minimum_required(VERSION 3.20)
project(dlks_formal LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(dlks_formal INTERFACE)
target_include_directories(dlks_formal INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)

include(CTest)
if(BUILD_TESTING)
  add_executable(test_quant tests/test_quant.cpp)
  target_link_libraries(test_quant PRIVATE dlks_formal)
  target_compile_options(test_quant PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME quant COMMAND test_quant)
endif()
