cmake_minimum_required(VERSION 3.20)
project(dlks_formal LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(dlks_formal INTERFACE)
target_include_directories(dlks_formal INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(dlks_formal INTERFACE cxx_std_20)

include(CTest)
if(BUILD_TESTING)
  add_executable(test_external_sort tests/test_external_sort.cpp)
  target_link_libraries(test_external_sort PRIVATE dlks_formal)
  add_test(NAME external_sort COMMAND test_external_sort)
endif()
