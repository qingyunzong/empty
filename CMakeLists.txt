cmake_minimum_required(VERSION 3.20)
project(linklib CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(linklib src/linker.cpp src/interpreter.cpp)
target_include_directories(linklib PUBLIC include)
if(MSVC)
    target_compile_options(linklib PRIVATE /W4)
else()
    target_compile_options(linklib PRIVATE -Wall -Wextra -Wpedantic)
endif()

enable_testing()
foreach(t basic conflicts weak gc circular oracle)
    add_executable(test_${t} tests/test_${t}.cpp)
    target_link_libraries(test_${t} PRIVATE linklib)
    target_include_directories(test_${t} PRIVATE tests)
    add_test(NAME ${t} COMMAND test_${t})
endforeach()
