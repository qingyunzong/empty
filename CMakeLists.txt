cmake_minimum_required(VERSION 3.16)
project(dataeng LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(dataeng dataeng.cpp)
target_include_directories(dataeng PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
enable_testing()

add_executable(test_bitemporal tests/test_bitemporal.cpp)
target_link_libraries(test_bitemporal PRIVATE dataeng)
add_test(NAME bitemporal COMMAND test_bitemporal)
