cmake_minimum_required(VERSION 3.16)
project(dataeng LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(dataeng dataeng.cpp)
target_include_directories(dataeng PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(dataeng PUBLIC cxx_std_20)
enable_testing()

if (CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
    add_executable(dataeng_deframer_test tests/deframer_test.cpp)
    target_link_libraries(dataeng_deframer_test PRIVATE dataeng)
    add_test(NAME dataeng_deframer_test COMMAND dataeng_deframer_test)
endif()
