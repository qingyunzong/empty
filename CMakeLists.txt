cmake_minimum_required(VERSION 3.16)
project(dataeng LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(dataeng dataeng.cpp)
target_include_directories(dataeng PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
enable_testing()

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/selftest.cpp")
  add_executable(dataeng_selftest tests/selftest.cpp)
  target_link_libraries(dataeng_selftest PRIVATE dataeng)
  add_test(NAME dataeng_selftest COMMAND dataeng_selftest)
endif()
