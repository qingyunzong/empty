cmake_minimum_required(VERSION 3.16)
project(dataeng LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(dataeng dataeng.cpp)
target_include_directories(dataeng PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(test_dataeng test_dataeng.cpp)
target_link_libraries(test_dataeng PRIVATE dataeng)
enable_testing()
add_test(NAME dataeng COMMAND test_dataeng)
